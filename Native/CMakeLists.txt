cmake_minimum_required(VERSION 3.20)
project(LocalVoiceWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
if(VOICE_ENGINE STREQUAL "speech")
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ENGINE_SOURCE}" engine)
  add_executable(localvoice-speech speech.cpp)
  target_link_libraries(localvoice-speech PRIVATE whisper)
  target_include_directories(localvoice-speech PRIVATE "${JSON_INCLUDE}")
else()
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ENGINE_SOURCE}" engine)
  add_executable(localvoice-editor editor.cpp)
  target_link_libraries(localvoice-editor PRIVATE llama)
  target_include_directories(localvoice-editor PRIVATE "${JSON_INCLUDE}")
endif()
