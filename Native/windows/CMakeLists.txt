cmake_minimum_required(VERSION 3.20)
project(VoiceWindowsDiarizer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build the Windows diarizer with Visual Studio 2022 x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(localvoice-diarizer ../diarize.cpp)
target_compile_definitions(localvoice-diarizer PRIVATE SHERPA_ONNX_BUILD_SHARED_LIBS NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(localvoice-diarizer PRIVATE /utf-8 /W4)
target_link_libraries(localvoice-diarizer PRIVATE "${SHERPA_LIBRARY_DIR}/sherpa-onnx-c-api.lib" advapi32)
